\chapter{Estado del Arte}
\label{cap:capitulo2}

\section{Tecnologías 5G}
\label{sec:5g}
\textit{Bandas FR1 (sub-6 GHz) vs FR2 (mmWave), arquitectura NR con gNodeB, NSA vs SA, KPIs (RSRP, SINR, throughput).}

\section{Burbujas Tácticas 5G}
\label{sec:burbujas}
\textit{Definición: celdas temporales para emergencias/eventos, casos de uso (terremotos, conciertos), equipos comerciales (Ericsson, Huawei).}

\section{Modelos de Propagación Radioeléctrica}
\label{sec:propagacion}
\textit{Free Space Path Loss (baseline teórico), Okumura-Hata (urbano), COST-231 (extensión Hata), Longley-Rice (elegido por terreno irregular), 3GPP TR 38.901 (estándar 5G).}

\section{Herramientas de Planificación Radioeléctrica}
\label{sec:herramientas}
\textit{Software comercial: Xirio Online (ray-tracing, €5k/año), Atoll (Forsk, industria), RadioPlanner (Siradel).\\
Alternativas open-source: SPLAT! (Linux, básico), CloudRF (freemium web).\\
Gap identificado: no existen herramientas gratuitas con interfaz moderna para 5G.}

\section{Conceptos de Gemelo Digital}
\label{sec:gemelo}
\textit{Definición: réplica virtual sincronizada con sistema físico, tipos (descriptivo, predictivo, prescriptivo), aplicaciones en telecom (optimización SON, gemelos de red).}

\section{Trabajos Relacionados}
\label{sec:relacionados}
\textit{Revisión de papers IEEE sobre planificación 5G, herramientas de investigación académica, diferenciación del TFG (foco en UX web, integración MATLAB-Spring Boot-Angular).}
